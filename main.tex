\section{Projects}
\project{Relay Doctor -- Agent Relay Diagnostics CLI}{TypeScript, Node.js, Vitest, JSON}
\begin{itemize}
  \item Built a command-line diagnostic tool detecting 4 classes of Agent Relay configuration drift -- missing node
        enrolment pins, legacy directories, cross-workspace conflicts and incomplete credentials.
  \item Implemented log diagnosis for 8 known failure signatures with root-cause guidance and PASS/\allowbreak WARN/\allowbreak FAIL/\allowbreak UNKNOWN
        classification, returning UNKNOWN rather than guessing on unrecognised input.
  \item Wrote 42 automated Vitest tests covering drift checks, log diagnosis, CLI behaviour and hardening, with
        redaction of 6 token formats and sensitive JSON fields.
\end{itemize}

\project{Retail Demand Forecasting and Inventory Planning System}{Python, Pandas, scikit-learn, statsmodels, SQL, FastAPI}
\begin{itemize}
  \item Built a forecasting pipeline on historical retail sales data, identifying demand patterns to drive
        stock-planning decisions.
  \item Prepared and validated datasets, then benchmarked statistical and machine-learning forecasting models
        using forecast-error measures.
  \item Developed a FastAPI service exposing forecasts to a dashboard, giving clear visibility of trends and predictions.
\end{itemize}

\project{Autonomous arXiv Paper Digest \& QA Agent}{Python, LLMs, retrieval-augmented generation (RAG), PDF processing}

        in source documents.
  \item Designed document ingestion, retrieval and prompt orchestration to improve traceability and reduce
        unsupported responses.
\end{itemize}

\project{Fleo -- Flight Deal Alert Application}{TypeScript, Next.js, React, PostgreSQL, Drizzle ORM, Supabase, Vitest}

  \item Implemented server-side free/premium gating, magic-link authentication and a rate-limited flight-data API
        client with retry backoff, backed by 66 Vitest tests.
